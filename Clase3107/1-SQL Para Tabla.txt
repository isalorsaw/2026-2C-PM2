CREATE TABLE `tbl_usuario_qr` (
	`id` INT(11) NOT NULL AUTO_INCREMENT,
	`usuario_id` INT(11) NOT NULL,
	`contenido` TEXT NOT NULL COLLATE 'utf8mb4_general_ci',
	`ruta_imagen` VARCHAR(255) NOT NULL COLLATE 'utf8mb4_general_ci',
	`tipo` VARCHAR(30) NULL DEFAULT 'registro' COLLATE 'utf8mb4_general_ci',
	`usado` TINYINT(1) NULL DEFAULT '0',
	`creado_en` DATETIME NULL DEFAULT CURRENT_TIMESTAMP,
	`pin` VARCHAR(4) NULL DEFAULT NULL COLLATE 'utf8mb4_general_ci',
	PRIMARY KEY (`id`) USING BTREE,
	UNIQUE INDEX `usuario_id` (`usuario_id`) USING BTREE,
	CONSTRAINT `tbl_usuario_qr_ibfk_1` FOREIGN KEY (`usuario_id`) REFERENCES `tbl_usuario` (`usuario_id`) ON UPDATE RESTRICT ON DELETE CASCADE
)
COLLATE='utf8mb4_general_ci'
ENGINE=InnoDB
AUTO_INCREMENT=13
;
